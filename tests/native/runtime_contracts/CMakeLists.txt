cmake_minimum_required(VERSION 3.18)
project(InfiniOpsRuntimeContracts LANGUAGES CXX)
option(WITH_HYGON "Include Hygon scalar Mul checks" OFF)
if(NOT "$ENV{INFINILM_CXX11_ABI}" MATCHES "^[01]$")
    message(FATAL_ERROR "Set INFINILM_CXX11_ABI to match the installed SDK")
endif()
foreach(name scalar_mul_regression cpu_cast_regression)
    add_executable(${name} ${name}.cpp)
    target_compile_features(${name} PRIVATE cxx_std_17)
    target_include_directories(${name} PRIVATE "$ENV{INFINI_ROOT}/include")
    target_compile_definitions(${name} PRIVATE _GLIBCXX_USE_CXX11_ABI=$ENV{INFINILM_CXX11_ABI})
    target_link_libraries(${name} PRIVATE "$ENV{INFINI_ROOT}/lib/libinfinirt.so" "$ENV{INFINI_ROOT}/lib/libinfiniops.so")
endforeach()
if(WITH_HYGON)
    find_package(CUDAToolkit REQUIRED)
    target_compile_definitions(scalar_mul_regression PRIVATE WITH_HYGON=1)
    target_link_libraries(scalar_mul_regression PRIVATE CUDA::cudart)
endif()
enable_testing()
add_test(NAME scalar_mul_regression COMMAND scalar_mul_regression)
add_test(NAME cpu_cast_regression COMMAND cpu_cast_regression)
