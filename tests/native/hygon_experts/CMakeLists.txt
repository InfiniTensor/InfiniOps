cmake_minimum_required(VERSION 3.18)
project(HygonExpertChecks LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
if(NOT "$ENV{INFINILM_CXX11_ABI}" MATCHES "^[01]$")
  message(FATAL_ERROR "Set INFINILM_CXX11_ABI to match the installed SDK")
endif()
add_library(expert_probe SHARED expert_probe.cu)
target_compile_features(expert_probe PRIVATE cxx_std_17 cuda_std_17)
target_include_directories(expert_probe PRIVATE ../../../src "$ENV{INFINI_ROOT}/include")
target_compile_definitions(expert_probe PRIVATE WITH_HYGON=1 _GLIBCXX_USE_CXX11_ABI=$ENV{INFINILM_CXX11_ABI})
target_link_libraries(expert_probe PRIVATE CUDA::cudart "$ENV{INFINI_ROOT}/lib/libinfinirt.so" "$ENV{INFINI_ROOT}/lib/libinfiniops.so")
